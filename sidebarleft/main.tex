\newcommand{\mpwidth}{\linewidth-\fboxsep-\fboxsep}

\newcommand{\cvsection}[1]{
\vspace{12pt}
{\LARGE\textbf{\textcolor{darkcol}{\MakeUppercase{#1}}}}\\[-4pt]
\textcolor{maincol}{\rule{0.1\textwidth}{2pt}}\\
}

\newcommand{\cvtext}[1]{
\begin{tabular*}{\mpwidth}{p{0.98\mpwidth}}
#1

}

\newcommand{\cvlist}[1]{\begin{itemize}#1\end{itemize}}

%----------------------------------------------------------------------------------------
%	НАВЫКИ (устойчивые полоски)
%----------------------------------------------------------------------------------------
\newcommand{\cvskill}[3]{
  \begin{tabular*}{\mpwidth}{p{0.72\mpwidth} r}
    \textbf{#1} & \textcolor{maincol}{#2}
  \end{tabular*}

  \pgfmathsetlengthmacro{\barW}{\mpwidth}
  \pgfmathsetlengthmacro{\fillW}{#3*\mpwidth}

  \begin{tikzpicture}[rounded corners=2pt,very thin]
    \fill [lightcol] (0,0) rectangle (\barW,0.15);
    \fill [maincol] (0,0) rectangle (\fillW,0.15);
  \end{tikzpicture}
}

\newcommand{\cvevent}[5]{
  \parbox{\mpwidth}{
    \begin{tabular*}{\mpwidth}{p{0.70\mpwidth} r}
      \textbf{#2} &
      \colorbox{maincol}{\makebox[0.34\mpwidth][c]{\textcolor{white}{#1}}} \\
      \textcolor{maincol}{#3} & \\
    \end{tabular*}\\[6pt]
    #4\\
  }
  #5
  \vspace{12pt}
}
